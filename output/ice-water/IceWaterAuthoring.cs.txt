// One-off authoring tool for the ice-on-water field (removed after use).
using System.Linq;
using System.Text;
using Hakaton.Expeditions;
using Hakaton.Feel;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;

static class IceWaterAuthoring
{
    const string Folder = "Assets/Expedition/Prefabs/Ice";
    const string BlockPath = Folder + "/Ice block.prefab";
    const string FieldPath = Folder + "/Ice field.prefab";

    public static string Run()
    {
        var log = new StringBuilder();
        if (!AssetDatabase.IsValidFolder(Folder)) AssetDatabase.CreateFolder("Assets/Expedition/Prefabs", "Ice");

        // Block prefab
        var blockGo = new GameObject("Ice block");
        var filter = blockGo.AddComponent<MeshFilter>();
        var renderer = blockGo.AddComponent<MeshRenderer>();
        renderer.sharedMaterial = AssetDatabase.LoadAssetAtPath<Material>("Assets/Expedition/Materials/Ice.mat");
        renderer.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.Off;
        var collider = blockGo.AddComponent<MeshCollider>();
        var block = blockGo.AddComponent<IceFieldBlock>();
        block.filter = filter; block.meshRenderer = renderer; block.meshCollider = collider;
        var blockPrefab = PrefabUtility.SaveAsPrefabAsset(blockGo, BlockPath).GetComponent<IceFieldBlock>();
        Object.DestroyImmediate(blockGo);
        log.AppendLine("block " + BlockPath);

        // Field prefab
        var fieldGo = new GameObject("Ice field");
        var field = fieldGo.AddComponent<IceField>();
        field.blockPrefab = blockPrefab;
        field.freezeBurst = Load<VfxBurst>("Assets/Game/Prefabs/Effects/Retro/Frost explosion.prefab");
        field.rimFrost = field.freezeBurst;
        field.meltSteam = Load<VfxBurst>("Assets/Game/Prefabs/Cards/TarPuddle/Tar steam.prefab");
        field.spreadSound = AssetDatabase.LoadAssetAtPath<SfxCue>("Assets/Audio/Cues/ice_spread.asset");
        field.settleSound = AssetDatabase.LoadAssetAtPath<SfxCue>("Assets/Audio/Cues/ice_freeze.asset");
        field.meltSound = AssetDatabase.LoadAssetAtPath<SfxCue>("Assets/Audio/Cues/fire_burst.asset");
        var fieldPrefab = PrefabUtility.SaveAsPrefabAsset(fieldGo, FieldPath).GetComponent<IceField>();
        Object.DestroyImmediate(fieldGo);
        log.AppendLine($"field {FieldPath} burst={field.freezeBurst} steam={field.meltSteam} spread={field.spreadSound} settle={field.settleSound} melt={field.meltSound}");

        // Prefab that carries an expedition (the enemy debug session)
        const string session = "Assets/EnemyDebug/Prefabs/Debug Session.prefab";
        var root = PrefabUtility.LoadPrefabContents(session);
        foreach (var world in root.GetComponentsInChildren<ExpeditionWorld>(true)) log.AppendLine(Wire(world, fieldPrefab, root.transform));
        PrefabUtility.SaveAsPrefabAsset(root, session);
        PrefabUtility.UnloadPrefabContents(root);

        string active = EditorSceneManager.GetActiveScene().path;
        foreach (var path in new[] { "Assets/Game/Scenes/Hakaton.unity", "Assets/Expedition/Scenes/BeastBrawlLab.unity", "Assets/ChestArt/Scenes/MimicLab.unity", "Assets/OddityArt/Scenes/OddityLab.unity" })
        {
            var scene = EditorSceneManager.OpenScene(path, OpenSceneMode.Single);
            foreach (var world in Object.FindObjectsByType<ExpeditionWorld>(FindObjectsInactive.Include, FindObjectsSortMode.None))
            {
                if (PrefabUtility.IsPartOfPrefabInstance(world) && world.ice) { log.AppendLine(path + ": in prefab, already wired"); continue; }
                if (PrefabUtility.IsPartOfPrefabInstance(world)) { log.AppendLine(path + ": WARNING world inside prefab instance " + world.name); continue; }
                log.AppendLine(path + ": " + Wire(world, fieldPrefab, null));
            }
            EditorSceneManager.MarkSceneDirty(scene);
            EditorSceneManager.SaveScene(scene);
        }

        // The cave's flooded stretch must be crossable with the one Ice card the tutorial hands over.
        {
            var scene = EditorSceneManager.OpenScene("Assets/TutorialCave/Scenes/TutorialCave.unity", OpenSceneMode.Single);
            foreach (var o in Object.FindObjectsByType<WorldObstacle>(FindObjectsInactive.Include, FindObjectsSortMode.None))
                if (o.kind == ObstacleKind.Lake)
                {
                    Undo.RecordObject(o, "ice radius");
                    o.minimumIceRadius = 6.5f;
                    EditorUtility.SetDirty(o);
                    log.AppendLine("cave lake " + o.name + " bounds " + o.SurfaceBounds.size + " surface " + o.HasSurface);
                }
            EditorSceneManager.MarkSceneDirty(scene);
            EditorSceneManager.SaveScene(scene);
        }
        if (!string.IsNullOrEmpty(active)) EditorSceneManager.OpenScene(active, OpenSceneMode.Single);
        AssetDatabase.SaveAssets();
        return log.ToString().Replace("\n", " | ");
    }

    static string Wire(ExpeditionWorld world, IceField prefab, Transform parent)
    {
        if (world.ice) return world.name + " already has " + world.ice.name;
        var field = (IceField)PrefabUtility.InstantiatePrefab(prefab, world.gameObject.scene);
        if (parent) field.transform.SetParent(parent, false);
        field.transform.SetPositionAndRotation(Vector3.zero, Quaternion.identity);
        field.transform.localScale = Vector3.one;
        field.world = world;
        world.ice = field;
        EditorUtility.SetDirty(field);
        EditorUtility.SetDirty(world);
        PrefabUtility.RecordPrefabInstancePropertyModifications(field);
        PrefabUtility.RecordPrefabInstancePropertyModifications(world);
        return "wired " + world.name;
    }

    static T Load<T>(string path) where T : Component
    {
        var go = AssetDatabase.LoadAssetAtPath<GameObject>(path);
        return go ? go.GetComponent<T>() : null;
    }

    /// <summary>Report the lakes in the open scene: surface bounds and whether ice can form.</summary>
    public static string Lakes()
    {
        var sb = new StringBuilder();
        foreach (var o in Object.FindObjectsByType<WorldObstacle>(FindObjectsInactive.Include, FindObjectsSortMode.None).Where(o => o.kind == ObstacleKind.Lake))
            sb.Append(o.name).Append(' ').Append(o.HasSurface).Append(' ').Append(o.SurfaceBounds.size).Append(" | ");
        return sb.ToString();
    }
}
